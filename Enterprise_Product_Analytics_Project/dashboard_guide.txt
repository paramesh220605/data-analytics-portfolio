DATA MODEL
Customer Master (1) -> (*) Sales Transactions (*) <- (1) Product Master
DateTable (1) -> (*) Sales Transactions

DAX:
DateTable = CALENDAR(MIN(Sales_Transactions[Order_Date]),MAX(Sales_Transactions[Order_Date]))
Year = YEAR(DateTable[Date])
Month = FORMAT(DateTable[Date],"MMM")
Month Number = MONTH(DateTable[Date])
Year Month = FORMAT(DateTable[Date],"YYYY-MM")

Total Revenue = SUM(Sales_Transactions[Revenue])
Total Cost = SUM(Sales_Transactions[Total_Cost])
Total Profit = SUM(Sales_Transactions[Profit])
Total Orders = DISTINCTCOUNT(Sales_Transactions[Order_ID])
Total Customers = DISTINCTCOUNT(Sales_Transactions[Customer_ID])
Units Sold = SUM(Sales_Transactions[Quantity])
Profit Margin % = DIVIDE([Total Profit],[Total Revenue],0)
Average Order Value = DIVIDE([Total Revenue],[Total Orders],0)
Revenue per Customer = DIVIDE([Total Revenue],[Total Customers],0)

Repeat Customers =
CALCULATE(
 DISTINCTCOUNT(Sales_Transactions[Customer_ID]),
 FILTER(VALUES(Sales_Transactions[Customer_ID]),
 CALCULATE(DISTINCTCOUNT(Sales_Transactions[Order_ID]))>1)
)

Repeat Customer Rate % = DIVIDE([Repeat Customers],[Total Customers],0)
